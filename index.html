Integrasi Google Apps Script Backend

Hubungkan SIMAS ke Google Sheets secara otomatis

Masukkan Web App URL yang disebarkan dari Google Spreadsheet Apps Script.

Langkah Integrasi Backend GAS:
  1. Buat Google Sheets baru di Google Drive Anda.
  2. Buka menu Extensi > Apps Script, tempelkan kode di bawah ini.
  3. Klik Deploy > New Deployment sebagai Web App (Access: Anyone), salin URL-nya.
Kode Google Apps Script (Code.gs): 
function doGet(e) {
  var action = e.parameter.action;
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  if (action === 'getAllData') {
    var profileSheet = getOrCreateSheet(ss, 'Profil');
    var pegawaiSheet = getOrCreateSheet(ss, 'Pegawai');
    var suratMasukSheet = getOrCreateSheet(ss, 'SuratMasuk');
    var arsipSheet = getOrCreateSheet(ss, 'ArsipDigital');
    
    return createJsonResponse({
      profil: getSheetData(profileSheet),
      pegawai: getSheetData(pegawaiSheet),
      suratMasuk: getSheetData(suratMasukSheet),
      arsip: getSheetData(arsipSheet)
    });
  }
  return createJsonResponse({ status: 'ok' });
}

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var action = data.action;
    
    if (action === 'saveProfil') {
      var sheet = getOrCreateSheet(ss, 'Profil');
      sheet.clear();
      sheet.appendRow(['key', 'value']);
      for (var k in data.payload) {
        sheet.appendRow([k, data.payload[k]]);
      }
    } else if (action === 'savePegawai') {
      saveArrayData(ss, 'Pegawai', ['id', 'nama', 'nip', 'nik', 'jabatan'], data.payload);
    } else if (action === 'saveSuratMasuk') {
      saveArrayData(ss, 'SuratMasuk', ['id', 'noBerkas', 'tglDiterima', 'nomorSurat', 'pengirim', 'perihal', 'fileName', 'fileData'], data.payload);
    } else if (action === 'saveArsip') {
      saveArrayData(ss, 'ArsipDigital', ['id', 'nomorSurat', 'jenisSurat', 'penerima', 'tanggal'], data.payload);
    }
    
    return createJsonResponse({ success: true });
  } catch(err) {
    return createJsonResponse({ success: false, error: err.toString() });
  }
}

function getOrCreateSheet(ss, name) {
  var sheet = ss.getSheetByName(name);
  if (!sheet) sheet = ss.insertSheet(name);
  return sheet;
}

function getSheetData(sheet) {
  var values = sheet.getDataRange().getValues();
  if (values.length <= 1) return [];
  var headers = values[0];
  var results = [];
  for (var i = 1; i < values.length; i++) {
    var obj = {};
    for (var j = 0; j < headers.length; j++) {
      obj[headers[j]] = values[i][j];
    }
    results.push(obj);
  }
  return results;
}

function saveArrayData(ss, sheetName, headers, items) {
  var sheet = getOrCreateSheet(ss, sheetName);
  sheet.clear();
  sheet.appendRow(headers);
  items.forEach(function(item) {
    var row = [];
    headers.forEach(function(h) { row.push(item[h] || ''); });
    sheet.appendRow(row);
  });
}

function createJsonResponse(data) {
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}
